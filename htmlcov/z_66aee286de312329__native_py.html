14def _oriented_rings(polygons: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

15 """ 

16 Rings of valid polygons oriented with the polygon interior on their left (exterior CCW, holes CW), 

17 with the position of the polygon each ring belongs to. 

18 """ 

19 if polygons.size == 0: 

20 return np.empty(0, dtype=object), np.empty(0, dtype=np.int64) 

21 rings, owner = shapely.get_rings(polygons, return_index=True) 

22 is_exterior = np.r_[True, owner[1:] != owner[:-1]] 

23 ccw = shapely.is_ccw(rings) 

24 flip = (is_exterior & ~ccw) | (~is_exterior & ccw) 

25 if flip.any(): 

26 rings[flip] = shapely.reverse(rings[flip]) 

27 return rings, owner

30def _lines_to_segments(lines: np.ndarray, owner: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

31 if lines.size == 0: 

32 return np.empty((0, 4), dtype="float64"), np.empty(0, dtype=np.int64) 

33 coords, index = shapely.get_coordinates(lines, return_index=True) 

34 same_line = index[1:] == index[:-1] 

35 segments = np.hstack([coords[:-1][same_line], coords[1:][same_line]]).astype("float64", copy=False) 

36 return segments, owner[index[:-1][same_line]]

39def geometries_to_segments(geometries, return_owner: bool = False): 

40 """ 

41 Convert obstacle geometries to wall segments for the Rust kernel. 

42 

43 Polygons contribute every ring (exterior and holes); lines contribute their own segments. 

44 Points and empty or missing geometries are ignored. 

45 

46 Returns: 

47 ``(segments, one_sided)``: an ``(m, 4)`` float64 array ``x1, y1, x2, y2`` and an ``(m,)`` bool 

48 array. ``True`` marks walls of valid polygons, oriented with the interior on the left, which 

49 the kernel may skip when the observer is on their interior side (back-face culling). Walls of 

50 lines and of invalid polygons stay two-sided. With ``return_owner=True`` a third ``(m,)`` array 

51 gives the position of the input geometry each wall comes from. 

52 """ 

53 empty = np.empty((0, 4), dtype="float64"), np.empty(0, dtype=bool), np.empty(0, dtype=np.int64) 

54 geoms = np.asarray(geometries, dtype=object) 

55 keep = ~shapely.is_missing(geoms) & ~shapely.is_empty(geoms) if geoms.size else np.zeros(0, dtype=bool) 

56 geom_positions = np.nonzero(keep)[0] 

57 parts, part_owner = shapely.get_parts(geoms[keep], return_index=True) if keep.any() else (np.empty(0), []) 

58 if len(parts) == 0: 

59 return empty if return_owner else empty[:2] 

60 part_owner = geom_positions[np.asarray(part_owner, dtype=np.int64)] 

61 type_ids = shapely.get_type_id(parts) 

62 is_polygon = type_ids == POLYGON_TYPE_ID 

63 polygons, polygon_owner = parts[is_polygon], part_owner[is_polygon] 

64 valid = shapely.is_valid(polygons) 

65 

66 rings, ring_polygon = _oriented_rings(polygons[valid]) 

67 one_sided, one_owner = _lines_to_segments(rings, polygon_owner[valid][ring_polygon]) 

68 

69 invalid_rings, invalid_ring_polygon = ( 

70 shapely.get_rings(polygons[~valid], return_index=True) if (~valid).any() else (np.empty(0), []) 

71 ) 

72 is_line = np.isin(type_ids, LINEAR_TYPE_IDS) 

73 two_lines = np.concatenate([np.asarray(invalid_rings, dtype=object), parts[is_line]]) 

74 two_owner = np.concatenate( 

75 [polygon_owner[~valid][np.asarray(invalid_ring_polygon, dtype=np.int64)], part_owner[is_line]] 

76 ) 

77 two_sided, two_owner = _lines_to_segments(two_lines, two_owner) 

78 

79 segments = np.vstack([one_sided, two_sided]) 

80 flags = np.r_[np.ones(len(one_sided), dtype=bool), np.zeros(len(two_sided), dtype=bool)] 

81 if return_owner: 

82 return segments, flags, np.r_[one_owner, two_owner].astype(np.int64) 

83 return segments, flags

86def native_visibility_polygons( 

87 points: np.ndarray, 

88 radii: np.ndarray, 

89 segments: np.ndarray, 

90 one_sided: np.ndarray | None = None, 

91 *, 

92 quad_segs: int = DEFAULT_QUAD_SEGS, 

93 n_threads: int | None = None, 

94) -> np.ndarray: 

95 """ 

96 Compute visibility polygons for many observers with the Rust kernel. 

97 

98 Args: 

99 points: ``(n, 2)`` observer coordinates in a projected CRS. 

100 radii: ``(n,)`` view radii in CRS units. 

101 segments: ``(m, 4)`` wall segments from :func:`geometries_to_segments`. 

102 one_sided: ``(m,)`` back-face culling flags from :func:`geometries_to_segments`. 

103 quad_segs: segments per quarter circle of the view disc (matches ``Point.buffer``). 

104 n_threads: worker threads, ``None`` for all cores. 

105 

106 Returns: 

107 Object array of ``n`` shapely Polygons (empty Polygon when nothing is visible). 

108 """ 

109 points = np.ascontiguousarray(points, dtype="float64").reshape(-1, 2) 

110 radii = np.ascontiguousarray(radii, dtype="float64").reshape(-1) 

111 segments = np.ascontiguousarray(segments, dtype="float64").reshape(-1, 4) 

112 if one_sided is not None: 112 ↛ 114line 112 didn't jump to line 114 because the condition on line 112 was always true

113 one_sided = np.ascontiguousarray(one_sided, dtype=bool).reshape(-1) 

114 coords, offsets = _visibility_polygons(points, radii, segments, int(quad_segs), n_threads, one_sided) 

115 has_ring = np.diff(offsets) > 0 

116 ring_offsets = np.concatenate([[0], offsets[1:][has_ring]]).astype("int64") 

117 geom_offsets = np.concatenate([[0], np.cumsum(has_ring)]).astype("int64") 

118 result = shapely.from_ragged_array(shapely.GeometryType.POLYGON, coords, (ring_offsets, geom_offsets)) 

119 result = np.asarray(result, dtype=object) 

120 invalid = ~shapely.is_valid(result) 

121 if invalid.any(): 

122 result[invalid] = [_largest_part_at(poly, Point(p)) for poly, p in zip(result[invalid], points[invalid])] 

123 return result

126def _polygon_parts(geometry) -> list[Polygon]: 

127 """All non-empty polygons inside a geometry, flattening nested multi-parts and collections.""" 

128 parts, stack = [], [geometry] 

129 while stack: 

130 geom = stack.pop() 

131 if geom is None or geom.is_empty: 131 ↛ 132line 131 didn't jump to line 132 because the condition on line 131 was never true

132 continue 

133 if geom.geom_type == "Polygon": 

134 parts.append(geom) 

135 elif geom.geom_type in ("MultiPolygon", "GeometryCollection"): 

136 stack.extend(geom.geoms) 

137 return parts

140def _largest_part_at(polygon, point: Point): 

141 """ 

142 Repair a (rarely) invalid ring and keep the polygonal part around the observer. 

143 

144 Near-coincident walls from different obstacles (e.g. a building and an earlier visibility 

145 polygon cut by the same reflection notch) can leave a micro bow-tie in the ring; the repair 

146 removes it without changing the area beyond floating-point noise. 

147 """ 

148 parts = _polygon_parts(shapely.make_valid(polygon)) 

149 if not parts: 149 ↛ 150line 149 didn't jump to line 150 because the condition on line 149 was never true

150 return Polygon() 

151 parts = _polygon_parts(shapely.union_all(parts)) 

152 containing = [g for g in parts if g.distance(point) <= 1e-6] 

153 return max(containing or parts, key=lambda g: g.area)
