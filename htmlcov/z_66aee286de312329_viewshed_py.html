40 """ 

41 Compute 2.5D visibility: the ground (and optionally the roofs) each observer can see from its eye height, 

42 among buildings of known height and terrain. 

43 

44 Unlike :func:`get_visibility`, which treats every obstacle as infinitely high, here a low building or a 

45 hill hides only what lies below the line of sight: an observer sees over a garage onto the street behind

48 centre (at ``target_height`` above the ground) passes above every building wall it crosses and above the 

49 terrain. Visible cells are merged into one polygon per observer. The eye is ``observer_height`` above the 

50 terrain; observers standing inside a building footprint are placed on its roof.

52 Args:

293def _target_grid(scene: Scene, xy, radius, cell, target_height, include_roofs, limit=False): 

294 """Cell grid around the observers (aligned to multiples of ``cell``) with the absolute height of the 

295 target in every cell: the ground plus ``target_height``, the roof on footprints with ``include_roofs``, 

296 NaN on footprints otherwise. Also returns the coordinate shift and the kernel's scene arguments.""" 

297 reach = float(np.nanmax(radius)) 

298 x0 = np.floor((xy[:, 0].min() - reach) / cell) * cell 

299 y0 = np.floor((xy[:, 1].min() - reach) / cell) * cell 

300 nx = int(np.ceil((xy[:, 0].max() + reach - x0) / cell)) + 1 

301 ny = int(np.ceil((xy[:, 1].max() + reach - y0) / cell)) + 1 

302 if limit and nx * ny > MAX_GRID_CELLS: 302 ↛ 303line 302 didn't jump to line 303 because the condition on line 302 was never true

303 raise ValueError( 

304 f"The grid would have {nx * ny:,} cells (limit {MAX_GRID_CELLS:,}); increase cell_size or split the area." 

305 ) 

306 shift = np.array([x0, y0]) 

307 kernel = kernel_scene(scene, shift) 

308 xs = x0 + (np.arange(nx) + 0.5) * cell 

309 ys = y0 + (np.arange(ny) + 0.5) * cell 

310 if scene.terrain_grid is None: 

311 target_z = np.full((ny, nx), float(target_height)) 

312 else: 

313 centres = np.column_stack([np.tile(xs, ny), np.repeat(ys, nx)]) 

314 target_z = (scene.ground_at(centres) + target_height).reshape(ny, nx) 

315 if len(scene.blds): 

316 roof = scene.base + scene.building_height 

317 footprint = rasterize_footprints( 

318 kernel[0], scene.owner, np.ascontiguousarray(roof, dtype=float), (0.0, 0.0, float(cell), nx, ny) 

319 ) 

320 covered = footprint >= 0 

321 target_z[covered] = roof[footprint[covered]] if include_roofs else np.nan 

322 return (float(x0), float(y0), float(cell), nx, ny), np.ascontiguousarray(target_z), shift, kernel

325def _targets_in_disc(targets: np.ndarray, grid, xy, radius) -> int: 

326 x0, y0, cell, nx, ny = grid 

327 i0, i1 = max(int((xy[0] - radius - x0) / cell), 0), min(int((xy[0] + radius - x0) / cell) + 1, nx) 

328 j0, j1 = max(int((xy[1] - radius - y0) / cell), 0), min(int((xy[1] + radius - y0) / cell) + 1, ny) 

329 cx = x0 + (np.arange(i0, i1) + 0.5) * cell - xy[0] 

330 cy = y0 + (np.arange(j0, j1) + 0.5) * cell - xy[1] 

331 disc = cx[None, :] ** 2 + cy[:, None] ** 2 <= radius**2 

332 return int((disc & targets[j0:j1, i0:i1]).sum())
