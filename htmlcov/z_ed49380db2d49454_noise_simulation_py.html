547def _recursive_simulation_queue( 

548 task_queue: multiprocessing.Queue, dead_area_dict: dict, dead_area_r: int, use_parallel: bool 

549): 

550 results = [] 

551 total_tasks = task_queue.qsize() 

552 

553 with tqdm(total=total_tasks, desc="Simulating noise") as pbar: 

554 if use_parallel: 

555 executor_class = concurrent.futures.ProcessPoolExecutor() 

556 else: 

557 executor_class = concurrent.futures.ThreadPoolExecutor() 

558 with executor_class as executor: 

559 max_workers = executor._max_workers # pylint: disable=protected-access 

560 future_to_task = {} 

561 while True: 

562 while not task_queue.empty() and len(future_to_task) < max_workers: 

563 func, task, kwargs = task_queue.get_nowait() 

564 future = executor.submit(func, task, **kwargs) 

565 future_to_task[future] = kwargs["simulation_ind"] 

566 done, _ = concurrent.futures.wait(future_to_task.keys(), return_when=concurrent.futures.FIRST_COMPLETED) 

567 for future in done: 

568 simulation_ind = future_to_task.pop(future) 

569 result, new_tasks = future.result() 

570 if new_tasks: 

571 new_tasks_n = 0 

572 local_dead_area = dead_area_dict.get(simulation_ind)
