65 path. Trees will scatter and absorb sound waves.

67 Recommended values are between 2 and 16, with higher values providing more accurate simulation results.

338def _reflection_candidates(points, obstacles, obstacles_union, vis_poly, point_from, max_dist): 

339 """ 

340 Visibility-polygon vertices that touch an obstacle and can seed a reflection, with the absorption 

341 ratio of the first obstacle they touch and their distance from the current source. 

342 Returns ``None`` when the geometry operations fail. 

343 """ 

344 buffers = shapely.buffer(points, 1, quad_segs=1) 

345 left, right = _pairs_sorted(obstacles.geometry.values, buffers, "intersects") 

346 first = np.r_[True, left[1:] != left[:-1]] if len(left) else np.empty(0, dtype=bool) 

347 left, right = left[first], right[first] 

348 absorb = obstacles["absorb_ratio"].to_numpy(dtype=float)[right] 

349 has_ratio = ~np.isnan(absorb) 

350 left, absorb = left[has_ratio], absorb[has_ratio] 

351 nearby_poly = point_from.buffer(1.1, quad_segs=2) 

352 try: 

353 free = shapely.difference( 

354 shapely.difference(shapely.difference(buffers[left], vis_poly), obstacles_union), nearby_poly 

355 ) 

356 except GEOSException: 

357 return None 

358 keep = ~shapely.is_empty(free) & (shapely.area(free) >= 0.01) 

359 left, absorb = left[keep], absorb[keep] 

360 dist = shapely.distance(points[left], point_from) 

361 near = dist < max_dist - 5 

362 return {"point": points[left][near], "absorb_ratio": absorb[near], "dist": dist[near]}

568def _recursive_simulation_queue(task_queue: deque, dead_area_dict: dict, dead_area_r: int, use_parallel: bool): 

569 results = [] 

570 total_tasks = len(task_queue) 

571 

572 with tqdm(total=total_tasks, desc="Simulating noise", disable=not config.enable_tqdm_bar) as pbar: 

573 if use_parallel: 

574 executor_class = concurrent.futures.ProcessPoolExecutor 

575 else: 

576 executor_class = concurrent.futures.ThreadPoolExecutor 

577 with executor_class() as executor: 

578 # Tasks run in waves (one reflection generation at a time) and results are consumed in 

579 # submission order, not completion order. The dead-area filter below therefore sees the 

580 # same sequence on every run and on every machine, whatever the worker timing; this is 

581 # exactly the order a single worker would process the queue in. 

582 while task_queue: 

583 wave = list(task_queue) 

584 task_queue.clear() 

585 futures = [executor.submit(func, task, **kwargs) for func, task, kwargs in wave] 

586 for (_, _, kwargs), future in zip(wave, futures): 

587 simulation_ind = kwargs["simulation_ind"] 

588 result, new_tasks = future.result() 

589 if new_tasks: 

590 new_tasks_n = 0 

591 local_dead_area = dead_area_dict.get(simulation_ind) 

592 new_dead_area_points = [local_dead_area] 

593 for func, new_task, new_kwargs in new_tasks: 

594 new_point = new_task[0] 

595 if not local_dead_area.covers(new_point): 

596 task_queue.append((func, new_task, new_kwargs)) 

597 new_dead_area_points.append(new_point.buffer(dead_area_r, quad_segs=2)) 

598 new_tasks_n += 1 

599 dead_area_dict[simulation_ind] = unary_union(new_dead_area_points) 

600 total_tasks += new_tasks_n 

601 pbar.total = total_tasks 

602 pbar.refresh() 

603 results.append(result) 

604 pbar.update(1) 

605 return results
